# 功能：只用 Qt Core/Test 构建 MPC 示例和回归测试，适合独立 Release 验证。
cmake_minimum_required(VERSION 3.16)
project(LateralMpcExample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
add_subdirectory(../../src/control lateral_control)
add_executable(lateral_mpc_demo main.cpp SimulationPlant.h)
target_link_libraries(lateral_mpc_demo PRIVATE adasmaster_lateral)
add_executable(test_lateral_mpc ../../tests/TestLateralMpc.cpp)
target_link_libraries(test_lateral_mpc PRIVATE adasmaster_lateral Qt6::Test)
enable_testing()
add_test(NAME test_lateral_mpc COMMAND test_lateral_mpc)
add_test(NAME lateral_demo_straight COMMAND lateral_mpc_demo straight)
add_test(NAME lateral_demo_circle COMMAND lateral_mpc_demo circle)
add_test(NAME lateral_demo_lane_change COMMAND lateral_mpc_demo lane-change)
add_executable(test_mpc_simulation ../../tests/TestMpcSimulation.cpp)
target_link_libraries(test_mpc_simulation PRIVATE adasmaster_lateral Qt6::Test)
add_test(NAME test_mpc_simulation COMMAND test_mpc_simulation)
